import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    it = iter(input_data)
    m = int(next(it))
    n = int(next(it))
    
    grid = [[next(it) for _ in range(n)] for _ in range(m)]
    k = int(next(it))
    instructions = [next(it) for _ in range(k)]
    
    def apply_gravity(g, rows, cols):
        for c in range(cols):
            cnt = 0
            for r in range(rows):
                if g[r][c] == '*':
                    cnt += 1
            dot_count = rows - cnt
            for r in range(dot_count):
                g[r][c] = '.'
            for r in range(dot_count, rows):
                g[r][c] = '*'
        return g

    grid = apply_gravity(grid, m, n)
    
    for inst in instructions:
        if inst == 'right':
            grid = [list(x) for x in zip(*grid[::-1])]
            m, n = n, m
        else:
            grid = [list(x) for x in zip(*grid)][::-1]
            m, n = n, m
        grid = apply_gravity(grid, m, n)
        
    for row in grid:
        print(" ".join(row))

if __name__ == '__main__':
    solve()
